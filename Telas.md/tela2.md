    function aplicarFiltros() {
      const termo = inputBusca.value.toLowerCase().trim();
      const statusSel = selectStatus.value;
      const areaSel = selectArea.value;

      const rowsDesktop = document.querySelectorAll('#tabela-clientes-body tr.client-row');
      const cardsMobile = document.querySelectorAll('#lista-clientes-mobile .client-card-mobile');

      let visiveis = 0;

      function matchElement(el) {
        const tipo = el.getAttribute('data-tipo');
        const area = el.getAttribute('data-area') || '';
        const status = el.getAttribute('data-status') || '';
        const textoTotal = el.textContent.toLowerCase();

        const matchTipo = (currentTipoFilter === 'todos') || (tipo === currentTipoFilter);
        const matchTermo = !termo || textoTotal.includes(termo);
        const matchStatus = !statusSel || status.includes(statusSel);
        const matchArea = !areaSel || area.includes(areaSel);

        return matchTipo && matchTermo && matchStatus && matchArea;
      }

      rowsDesktop.forEach(row => {
        if (matchElement(row)) {
          row.classList.remove('hidden');
          visiveis++;
        } else {
          row.classList.add('hidden');
        }
      });

      cardsMobile.forEach(card => {
        if (matchElement(card)) {
          card.classList.remove('hidden');
        } else {
          card.classList.add('hidden');
        }
      });

      contagemResultados.textContent = visiveis + ' clientes listados';

      const hasActiveFilters = termo || statusSel || areaSel || currentTipoFilter !== 'todos';
      if (hasActiveFilters) {
        filtroBadge.classList.remove('hidden');
      } else {
        filtroBadge.classList.add('hidden');
      }
    }
