  function switchProfile(profile) {

    const submitBtn = document.getElementById('submit-btn');

    if (profile === 'lawyer') {

      input.placeholder = 'Ex: SP 418.992 ou 000.000.000-00';

      submitBtn.querySelector('span:nth-child(2)').innerText = 'Acessar Central do Cliente';
    }
  }

  function togglePasswordVisibility() {
    const pwd = document.getElementById('password-input');
    const icon = document.getElementById('password-toggle-icon');
    if (pwd.type === 'password') {
      pwd.type = 'text';
      icon.innerText = 'visibility_off';
    } else {
      pwd.type = 'password';
      icon.innerText = 'visibility';
    }
  }

  function showNotice(text, icon = 'info') {
    const notice = document.getElementById('authNotice');
    const noticeText = document.getElementById('authNoticeText');
    const noticeIcon = document.getElementById('authNoticeIcon');
    noticeText.innerText = text;
    noticeIcon.innerText = icon;
    notice.classList.remove('hidden');
    clearTimeout(window.noticeTimer);
    window.noticeTimer = setTimeout(() => {
      notice.classList.add('hidden');
    }, 4500);
  }

  function handleLoginSubmit(event) {
    event.preventDefault();
    const btn = document.getElementById('submit-btn');
    const originalText = btn.innerHTML;
    btn.innerHTML = '<span class="material-symbols-outlined animate-spin text-[20px]">progress_activity</span><span>Autenticando...</span>';
    btn.disabled = true;

    setTimeout(() => {
      btn.innerHTML = originalText;
      btn.disabled = false;
      showNotice('Autenticação segura realizada. Redirecionando para os processos...', 'check_circle');
    }, 1200);
  }

  function handleCertificateLogin() {
    showNotice('Buscando certificado digital A1/A3 ou leitor de token USB...', 'token');
  }

  function triggerBiometrics(method) {
    showNotice('Aproxime o sensor para validação por ' + method + '...', 'fingerprint');
  }
